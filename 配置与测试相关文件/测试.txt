					pData = recvData;
					checkSum = recvLength;

					{

						}
						else if (dataTemplates[recvData[0]].data_property.type == TYPE_TEMPLATE_BOOL)
						{
							*((unsigned char *)(dataTemplates[recvData[0]].data_property.data)) = (recvData[1] > 0) ? 1 : 0;
						}
					}
				}
				recvFlag = 0;
				break;
			}
		}
